# ListingLens Copilot — eval-only extras.
#
# These layer on top of requirements.txt; they are not a standalone set.
# Everything the app needs at runtime — including the agent layer
# (langgraph, mcp, instructor) — lives in requirements.txt and is installed
# by the Render build. This file holds only what the offline evaluation
# harness needs, which is why it stays out of the production install.
#
# Do not re-add runtime pins here. Duplicating a pin across both files is
# how the two environments drifted apart last time: requirements.txt gets
# bumped, this file doesn't, and Render ends up resolving something the
# eval job never saw.
#
# Used by:
#   - .github/workflows/eval-on-pr.yml (smoke eval on PRs)
#   - local eval runs:
#       uv venv .venv
#       source .venv/bin/activate
#       uv pip install -r requirements.txt -r requirements-agent.txt
#
# Both pins were validated to NOT downgrade the langchain v1 ecosystem
# pinned in requirements.txt.

# Evaluation harness (DeepEval handles LLM-as-judge primitives)
deepeval>=2.0

# Tracing / observability for LangGraph
