# Floor, not an exact pin: deepeval>=4 requires python-dotenv>=1.1.1, and
# deepeval 4 is required because its pricing table is the only one that knows
# claude-haiku-4-5 — the judge model eval/judges.py defaults to. On deepeval
# 2.9.3 the judge API call succeeds and then crashes in calculate_cost with
# KeyError: 'input', because model_pricing.get(name, model_pricing) falls back
# to the whole table for an unknown model. Nothing here needs 1.0.1 exactly.
python-dotenv>=1.1.1,<2
pydantic>=2.10,<3

duckdb>=1.0,<2

# RAG.
#
# No torch, transformers or sentence-transformers here — deliberately. The
# only thing this app ever used them for was MiniLM embeddings, and that
# import chain cost ~490MiB of RSS before reading a single weight, which is
# what pushed the RAG path over a 512MB instance. src/onnx_embeddings.py runs
# the same checkpoint through onnxruntime instead and produces identical
# vectors (cosine 1.000000 against the committed FAISS indexes, asserted by
# tests/test_onnx_embeddings.py). Measured in the production image:
# 705MiB -> 380MiB, and the image drops ~1.1GB (3.59GB -> 2.48GB).
#
# If you ever reintroduce sentence-transformers, pin it as
# `langchain-huggingface[full]` — the plain package declares no
# sentence-transformers floor at all, so pip will silently pair it with a
# 2.x install and report no conflict.
#
# faiss-cpu floor is 1.13 — the first release with cp313 wheels. Note this
# raises the local-dev floor to macOS 14: faiss-cpu >=1.13 publishes
# `macosx_14_0_arm64` wheels only. Linux (Render/Docker) is unaffected.
faiss-cpu>=1.13,<2
onnxruntime>=1.20,<2
tokenizers>=0.20,<1
huggingface-hub>=0.33,<2
langchain==1.2.13
langchain-core==1.2.21
langchain-community==0.4.1
langchain-groq==1.1.2
langchain-text-splitters==1.1.1
